    private static String join(List<String> list) {
        if (list.isEmpty()) return "без відключень";
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < list.size(); i++) {
            if (i > 0) sb.append(", ");
            sb.append(list.get(i));
        }
        return sb.toString();

package com.kiparisss.svitlo;

import java.util.ArrayList;
import java.util.Calendar;
import java.util.HashSet;
import java.util.LinkedHashSet;
import java.util.List;
import java.util.Locale;
import java.util.Set;
import java.util.TimeZone;
import java.util.TreeMap;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

/** Розбір сторінки графіків Житомиробленерго. Чиста Java, без Android. */
public final class Parser {
    public static final int SLOTS = 48;
    public static final TimeZone KYIV = TimeZone.getTimeZone("Europe/Kiev");

    public static final class Day {
        public String date;
        public boolean[] off;
    }

    public static final class Data {
        public final List<Day> days = new ArrayList<>();
        public String updated;
        public long fetchedAt;
    }

    public static final class Status {
        public boolean known;
        public boolean off;
        public long change = -1;   // -1: змін до кінця графіка немає
        public long graphEnd;
    }

    private Parser() {}

    public static int pidOf(String queue) {
        String[] p = queue.split("\\.");
        return (Integer.parseInt(p[0]) - 1) * 2 + Integer.parseInt(p[1]);
    }

    static boolean isReddish(int r, int g, int b) {
        return r >= 170 && g <= 110 && b <= 110;
    }

    static boolean colorIsRed(String s) {
        s = s.toLowerCase(Locale.ROOT);
        if (Pattern.compile("\\b(red|crimson|tomato|firebrick|darkred|orangered)\\b").matcher(s).find()) return true;
        Matcher m = Pattern.compile("#([0-9a-f]{6}|[0-9a-f]{3})\\b").matcher(s);
        while (m.find()) {
            String h = m.group(1);
            if (h.length() == 3) {
                h = "" + h.charAt(0) + h.charAt(0) + h.charAt(1) + h.charAt(1) + h.charAt(2) + h.charAt(2);
            }
            int r = Integer.parseInt(h.substring(0, 2), 16);
            int g = Integer.parseInt(h.substring(2, 4), 16);
            int b = Integer.parseInt(h.substring(4, 6), 16);
            if (isReddish(r, g, b)) return true;
        }
        m = Pattern.compile("rgba?\\(\\s*(\\d{1,3})\\s*,\\s*(\\d{1,3})\\s*,\\s*(\\d{1,3})").matcher(s);
        while (m.find()) {
            if (isReddish(Integer.parseInt(m.group(1)), Integer.parseInt(m.group(2)), Integer.parseInt(m.group(3)))) return true;
        }
        return false;
    }

    /** CSS-класи з блоків style, що фарбують фон у червоний. */
    static Set<String> redClasses(String html) {
        Set<String> set = new HashSet<>();
        Matcher style = Pattern.compile("<style[^>]*>([\\s\\S]*?)</style>", Pattern.CASE_INSENSITIVE).matcher(html);
        while (style.find()) {
            Matcher m = Pattern.compile("\\.([a-zA-Z0-9_-]+)[^{}]*\\{[^}]*background(?:-color)?\\s*:\\s*([^;}]+)").matcher(style.group(1));
            while (m.find()) {
                if (colorIsRed(m.group(2))) set.add(m.group(1).toLowerCase(Locale.ROOT));
            }
        }
        return set;
    }

    static boolean cellIsOff(String markup, Set<String> redClasses) {
        String s = markup.toLowerCase(Locale.ROOT);
        if (colorIsRed(s)) return true;
        Matcher m = Pattern.compile("class\\s*=\\s*[\"']?([^\"'>]+)").matcher(s);
        while (m.find()) {
            for (String name : m.group(1).trim().split("\\s+")) {
                if (redClasses.contains(name)) return true;
                if (name.matches(".*\\b(off|red|cherv|vidkl|danger).*")) return true;
            }
        }
        return false;
    }

    /** Ознака кольору клітинки: фон, bgcolor або клас (рамки не враховуються). */
    static String colorSig(String markup) {
        String s = markup.toLowerCase(Locale.ROOT);
        Matcher m = Pattern.compile("background(?:-color)?\\s*:\\s*([^;\"']+)").matcher(s);
        if (m.find()) return "bg:" + m.group(1).trim();
        m = Pattern.compile("bgcolor\\s*=\\s*[\"']?([^\"'\\s>]+)").matcher(s);
        if (m.find()) return "bgcolor:" + m.group(1);
        m = Pattern.compile("class\\s*=\\s*[\"']?([^\"'>]+)").matcher(s);
        if (m.find()) return "class:" + m.group(1).trim();
        return s.replaceAll("\\s+", " ").trim();
    }

    public static Data parse(String html, int pid) throws Exception {
        Set<String> redClasses = redClasses(html);
        Pattern cellRe = Pattern.compile(
                "<td([^>]*)>\\s*(?:<(?!/?td)[^>]+>\\s*)*<a[^>]*pidcherga_id=" + pid + "(?![0-9])[^>]*>([\\s\\S]*?)</a>",
                Pattern.CASE_INSENSITIVE);
        Matcher m = cellRe.matcher(html);
        List<Boolean> cells = new ArrayList<>();
        Set<String> sigs = new LinkedHashSet<>();
        while (m.find()) {
            String inner = m.group(2);
            String text = inner.replaceAll("<[^>]+>", "").replace("&nbsp;", " ").trim();
            if (text.matches("\\d\\.\\d")) continue; // підпис «3.1», а не часова клітинка
            String markup = m.group(1) + " " + inner.replaceAll("(?i)href\\s*=\\s*[\"'][^\"']*[\"']", "");
            cells.add(cellIsOff(markup, redClasses));
            sigs.add(colorSig(markup));
        }
        if (cells.size() < SLOTS) {
            throw new Exception("Не знайшов графік для підчерги (" + cells.size() + " клітинок). Можливо, сайт змінив розмітку.");
        }
        boolean anyOff = false;
        for (Boolean b : cells) anyOff |= b;
        if (!anyOff && sigs.size() > 1) {
            throw new Exception("Не вдалося розпізнати кольори графіка: " + sigs);
        }

        Data data = new Data();
        data.fetchedAt = System.currentTimeMillis();
        Matcher um = Pattern.compile("(\\d{1,2}:\\d{2})\\s+(\\d{2}\\.\\d{2}\\.\\d{4})").matcher(html);
        data.updated = um.find() ? um.group(0) : null;

        TreeMap<Long, String> dates = new TreeMap<>();
        Matcher dm = Pattern.compile("\\b\\d{2}\\.\\d{2}\\.\\d{4}\\b").matcher(html);
        while (dm.find()) {
            try { dates.put(dayStart(dm.group()), dm.group()); } catch (Exception ignored) { }
        }
        List<String> all = new ArrayList<>(dates.values());
        int dayCount = cells.size() / SLOTS;
        List<String> last = all.size() >= dayCount ? all.subList(all.size() - dayCount, all.size()) : all;

        for (int d = 0; d < dayCount; d++) {
            Day day = new Day();
            day.date = d < last.size() ? last.get(d) : null;
            day.off = new boolean[SLOTS];
            for (int i = 0; i < SLOTS; i++) day.off[i] = cells.get(d * SLOTS + i);
            data.days.add(day);
        }
        return data;
    }

    /* ───────── час ───────── */

    public static long dayStart(String date) {
        String[] p = date.split("\\.");
        Calendar c = Calendar.getInstance(KYIV);
        c.clear();
        c.set(Integer.parseInt(p[2]), Integer.parseInt(p[1]) - 1, Integer.parseInt(p[0]), 0, 0, 0);
        return c.getTimeInMillis();
    }

    static long slotStart(long dayStart, int i) {
        Calendar c = Calendar.getInstance(KYIV);
        c.setTimeInMillis(dayStart);
        c.add(Calendar.MINUTE, i * 30);
        return c.getTimeInMillis();
    }

    public static String dateStr(long t, int addDays) {
        Calendar c = Calendar.getInstance(KYIV);
        c.setTimeInMillis(t);
        c.add(Calendar.DAY_OF_MONTH, addDays);
        return String.format(Locale.ROOT, "%02d.%02d.%04d",
                c.get(Calendar.DAY_OF_MONTH), c.get(Calendar.MONTH) + 1, c.get(Calendar.YEAR));
    }

    public static String hhmm(long t) {
        Calendar c = Calendar.getInstance(KYIV);
        c.setTimeInMillis(t);
        return String.format(Locale.ROOT, "%02d:%02d", c.get(Calendar.HOUR_OF_DAY), c.get(Calendar.MINUTE));
    }

    /** Хвилини від початку доби за Києвом (для позначки «зараз» на смужці). */
    public static int minutesOfDay(long t) {
        Calendar c = Calendar.getInstance(KYIV);
        c.setTimeInMillis(t);
        return c.get(Calendar.HOUR_OF_DAY) * 60 + c.get(Calendar.MINUTE);
    }

    public static Status status(Data data, long now) {
        List<long[]> tl = new ArrayList<>(); // {start, off}
        for (int d = 0; d < data.days.size(); d++) {
            Day day = data.days.get(d);
            long base = day.date != null ? dayStart(day.date) : dayStart(dateStr(now, d));
            for (int i = 0; i < SLOTS; i++) tl.add(new long[]{slotStart(base, i), day.off[i] ? 1 : 0});
        }
        Status st = new Status();
        int idx = -1;
        for (int k = 0; k < tl.size(); k++) {
            long start = tl.get(k)[0];
            long end = k + 1 < tl.size() ? tl.get(k + 1)[0] : slotStart(start, 1);
            if (start <= now && now < end) { idx = k; break; }
        }
        if (idx < 0) return st;
        st.known = true;
        st.off = tl.get(idx)[1] == 1;
        int j = idx + 1;
        while (j < tl.size() && (tl.get(j)[1] == 1) == st.off) j++;
        st.change = j < tl.size() ? tl.get(j)[0] : -1;
        st.graphEnd = slotStart(tl.get(tl.size() - 1)[0], 1);
        return st;
    }

    public static boolean[] offFor(Data data, long now, int addDays) {
        String want = dateStr(now, addDays);
        for (Day d : data.days) if (want.equals(d.date)) return d.off;
        if (addDays == 0 && data.days.size() == 1 && data.days.get(0).date == null) return data.days.get(0).off;
        return null;
    }

    static String slotLabel(int i) {
        if (i >= SLOTS) return "24:00";
        return String.format(Locale.ROOT, "%02d:%s", i / 2, i % 2 == 1 ? "30" : "00");
    }

    public static List<String> intervals(boolean[] off) {
        List<String> res = new ArrayList<>();
        int s = -1;
        for (int i = 0; i < off.length; i++) {
            if (off[i] && s < 0) s = i;
            if (!off[i] && s >= 0) { res.add(slotLabel(s) + "–" + slotLabel(i)); s = -1; }
        }
        if (s >= 0) res.add(slotLabel(s) + "–" + slotLabel(off.length));
        return res;
    }

    /* ───────── кеш у вигляді рядка ───────── */

    public static String serialize(Data d) {
        StringBuilder sb = new StringBuilder();
        sb.append(d.updated == null ? "" : d.updated).append('\n').append(d.fetchedAt).append('\n');
        for (Day day : d.days) {
            sb.append(day.date == null ? "-" : day.date).append('|');
            for (boolean b : day.off) sb.append(b ? '1' : '0');
            sb.append('\n');
        }
        return sb.toString();
    }

    public static Data deserialize(String s) {
        if (s == null || s.isEmpty()) return null;
        try {
            String[] lines = s.split("\n");
            Data d = new Data();
            d.updated = lines[0].isEmpty() ? null : lines[0];
            d.fetchedAt = Long.parseLong(lines[1]);
            for (int k = 2; k < lines.length; k++) {
                String[] p = lines[k].split("\\|");
                if (p.length != 2 || p[1].length() != SLOTS) continue;
                Day day = new Day();
                day.date = "-".equals(p[0]) ? null : p[0];
                day.off = new boolean[SLOTS];
                for (int i = 0; i < SLOTS; i++) day.off[i] = p[1].charAt(i) == '1';
                d.days.add(day);
            }
            return d.days.isEmpty() ? null : d;
        } catch (Exception e) {
            return null;
        }
    }

import android.content.Context;
import android.content.SharedPreferences;
import android.graphics.Bitmap;
import android.graphics.Canvas;
import android.graphics.Paint;
import android.graphics.RectF;
import android.graphics.Typeface;

import java.io.ByteArrayOutputStream;
import java.io.InputStream;
import java.net.HttpURLConnection;
import java.net.URL;
import java.nio.charset.Charset;

/** Завантаження, кеш і спільні для застосунку та віджета речі. */
public final class Schedule {
    public static final String SOURCE_URL = "https://www.ztoe.com.ua/unhooking-search.php";
    private static final String PREFS = "svitlo";

    public static final int COLOR_ON = 0xFF4ADE80;
    public static final int COLOR_OFF = 0xFFFB4D6D;
    public static final int COLOR_DIM = 0xFF9B93B8;

    private Schedule() {}

    private static SharedPreferences prefs(Context c) {
        return c.getSharedPreferences(PREFS, Context.MODE_PRIVATE);
    }

    public static String getQueue(Context c) {
        return prefs(c).getString("queue", null);
    }

    public static void setQueue(Context c, String q) {
        prefs(c).edit().putString("queue", q).apply();
    }

    public static Parser.Data load(Context c, String q) {
        return Parser.deserialize(prefs(c).getString("data_" + q, null));
    }

    public static boolean isOffline(Context c, String q) {
        return prefs(c).getBoolean("offline_" + q, false);
    }

    /** Завантажує графік із сайту й зберігає. Кидає виняток, якщо не вдалося. */
    public static Parser.Data refresh(Context c, String q) throws Exception {
        try {
            Parser.Data d = Parser.parse(fetchHtml(), Parser.pidOf(q));
            prefs(c).edit()
                    .putString("data_" + q, Parser.serialize(d))
                    .putBoolean("offline_" + q, false)
                    .apply();
            return d;
        } catch (Exception e) {
            prefs(c).edit().putBoolean("offline_" + q, true).apply();
            throw e;
        }
    }

    static String fetchHtml() throws Exception {
        HttpURLConnection con = (HttpURLConnection) new URL(SOURCE_URL).openConnection();
        con.setConnectTimeout(15000);
        con.setReadTimeout(20000);
        con.setRequestProperty("User-Agent", "Mozilla/5.0 (Linux; Android) SvitloZhytomyr/1.0");
        try (InputStream in = con.getInputStream()) {
            ByteArrayOutputStream bo = new ByteArrayOutputStream();
            byte[] buf = new byte[16384];
            int n;
            while ((n = in.read(buf)) > 0) bo.write(buf, 0, n);
            return new String(bo.toByteArray(), Charset.forName("windows-1251"));
        } finally {
            con.disconnect();
        }
    }

    static String clock(long t) {
        String s = Parser.hhmm(t);
        return "00:00".equals(s) ? "24:00" : s;
    }

    /** [заголовок, рядок, колір] для поточного стану. */
    public static Object[] texts(Parser.Status st) {
        if (!st.known) return new Object[]{"Немає графіка", "на поточний час", COLOR_DIM};
        if (st.off) {
            String line = st.change >= 0 ? "Увімкнуть о " + clock(st.change) : "Без світла до " + clock(st.graphEnd);
            return new Object[]{"Світла немає", line, COLOR_OFF};
        }
        String line = st.change >= 0 ? "Вимкнуть о " + clock(st.change) : "Відключень не заплановано";
        return new Object[]{"Світло є", line, COLOR_ON};
    }

    /** Смужка на 24 години з позначкою поточного часу. */
    public static Bitmap bar(boolean[] off, long now, int width, boolean labels) {
        int barH = Math.round(width * 0.05f);
        int labelH = labels ? Math.round(width * 0.045f) : 0;
        Bitmap bmp = Bitmap.createBitmap(width, barH + labelH + 4, Bitmap.Config.ARGB_8888);
        Canvas c = new Canvas(bmp);
        Paint p = new Paint(Paint.ANTI_ALIAS_FLAG);
        float seg = width / (float) off.length;
        float gap = Math.max(1f, seg * 0.15f);
        for (int i = 0; i < off.length; i++) {
            p.setColor(off[i] ? COLOR_OFF : 0xFF2F5A3C);
            c.drawRoundRect(new RectF(i * seg + gap / 2, 2, (i + 1) * seg - gap / 2, barH + 2), 3, 3, p);
        }
        if (now > 0) {
            float x = Parser.minutesOfDay(now) / 1440f * width;
            p.setColor(0xFFFFFFFF);
            float w = Math.max(3f, width * 0.006f);
            c.drawRect(Math.max(0, x - w / 2), 0, Math.min(width, x + w / 2), barH + 4, p);
        }
        if (labels) {
            p.setColor(COLOR_DIM);
            p.setTextSize(labelH * 0.85f);
            p.setTypeface(Typeface.DEFAULT);
            float y = barH + 4 + labelH * 0.85f;
            String[] t = {"0", "6", "12", "18", "24"};
            for (int k = 0; k < t.length; k++) {
                float x = k * width / 4f;
                Paint.Align a = k == 0 ? Paint.Align.LEFT : k == 4 ? Paint.Align.RIGHT : Paint.Align.CENTER;
                p.setTextAlign(a);
                c.drawText(t[k], x, y, p);
            }
        }
        return bmp;
    }
}

package com.kiparisss.svitlo;

import android.app.AlarmManager;
import android.app.PendingIntent;
import android.appwidget.AppWidgetManager;
import android.appwidget.AppWidgetProvider;
import android.content.ComponentName;
import android.content.Context;
import android.content.Intent;
import android.os.SystemClock;
import android.view.View;
import android.widget.RemoteViews;

import java.util.List;

/** Віджет головного екрана. */
public class SvitloWidget extends AppWidgetProvider {
    public static final String ACTION_REFRESH = "com.kiparisss.svitlo.REFRESH";

    @Override
    public void onUpdate(Context ctx, AppWidgetManager mgr, int[] ids) {
        refreshAsync(ctx, goAsync());
    }

    @Override
    public void onReceive(Context ctx, Intent intent) {
        super.onReceive(ctx, intent);
        if (ACTION_REFRESH.equals(intent.getAction())) {
            refreshAsync(ctx, goAsync());
        }
    }

    private static void refreshAsync(final Context ctx, final PendingResult result) {
        final Context app = ctx.getApplicationContext();
        new Thread(() -> {
            try {
                String q = Schedule.getQueue(app);
                if (q != null) {
                    try { Schedule.refresh(app, q); } catch (Exception ignored) { }
                }
                updateAll(app);
            } finally {
                result.finish();
            }
        }).start();
    }

    /** Перемальовує всі віджети з кешу й планує наступне оновлення. */
    public static void updateAll(Context ctx) {
        AppWidgetManager mgr = AppWidgetManager.getInstance(ctx);
        int[] ids = mgr.getAppWidgetIds(new ComponentName(ctx, SvitloWidget.class));
        if (ids == null || ids.length == 0) return;
        long next = build(ctx, mgr, ids);
        scheduleNext(ctx, next);
    }

    /** Повертає час наступної зміни стану (або -1). */
    private static long build(Context ctx, AppWidgetManager mgr, int[] ids) {
        RemoteViews rv = new RemoteViews(ctx.getPackageName(), R.layout.widget);
        Intent open = new Intent(ctx, MainActivity.class);
        rv.setOnClickPendingIntent(R.id.root, PendingIntent.getActivity(ctx, 0, open,
                PendingIntent.FLAG_IMMUTABLE | PendingIntent.FLAG_UPDATE_CURRENT));

        long now = System.currentTimeMillis();
        String q = Schedule.getQueue(ctx);
        Parser.Data data = q == null ? null : Schedule.load(ctx, q);
        long change = -1;

        rv.setViewVisibility(R.id.offline, View.GONE);
        rv.setViewVisibility(R.id.timerRow, View.GONE);
        rv.setViewVisibility(R.id.bar, View.GONE);
        rv.setTextViewText(R.id.today, "");

        if (q == null) {
            rv.setTextViewText(R.id.queue, "Світло · Житомир");
            rv.setTextViewText(R.id.status, "Оберіть підчергу");
            rv.setTextColor(R.id.status, Schedule.COLOR_DIM);
            rv.setTextViewText(R.id.line, "Натисніть, щоб налаштувати");
            rv.setTextViewText(R.id.updated, "");
        } else if (data == null) {
            rv.setTextViewText(R.id.queue, "Черга " + q);
            rv.setTextViewText(R.id.status, "Немає даних");
            rv.setTextColor(R.id.status, Schedule.COLOR_DIM);
            rv.setTextViewText(R.id.line, "Перевірте інтернет");
            rv.setTextViewText(R.id.updated, "ztoe.com.ua");
        } else {
            Parser.Status st = Parser.status(data, now);
            Object[] t = Schedule.texts(st);
            rv.setTextViewText(R.id.queue, "Черга " + q);
            rv.setTextViewText(R.id.status, (String) t[0]);
            rv.setTextColor(R.id.status, (Integer) t[2]);
            rv.setTextViewText(R.id.line, (String) t[1]);
            if (Schedule.isOffline(ctx, q)) rv.setViewVisibility(R.id.offline, View.VISIBLE);

            if (st.known && st.change >= 0) {
                change = st.change;
                rv.setViewVisibility(R.id.timerRow, View.VISIBLE);
                long base = SystemClock.elapsedRealtime() + (st.change - now);
                rv.setChronometer(R.id.timer, base, null, true);
                rv.setChronometerCountDown(R.id.timer, true);
            }

            boolean[] today = Parser.offFor(data, now, 0);
            if (today != null) {
                rv.setViewVisibility(R.id.bar, View.VISIBLE);
                rv.setImageViewBitmap(R.id.bar, Schedule.bar(today, now, 600, true));
                List<String> list = Parser.intervals(today);
                StringBuilder sb = new StringBuilder("СЬОГОДНІ");
                if (list.isEmpty()) sb.append("\nбез відключень");
                for (int i = 0; i < Math.min(3, list.size()); i++) sb.append('\n').append(list.get(i));
                if (list.size() > 3) sb.append("\n…");
                rv.setTextViewText(R.id.today, sb.toString());
            }
            String upd = data.updated != null ? data.updated.split(" ")[0] : null;
            rv.setTextViewText(R.id.updated, upd != null ? "Оновлено " + upd + " · ztoe.com.ua" : "ztoe.com.ua");
        }
        mgr.updateAppWidget(ids, rv);
        return change;
    }

    private static void scheduleNext(Context ctx, long change) {
        AlarmManager am = (AlarmManager) ctx.getSystemService(Context.ALARM_SERVICE);
        if (am == null) return;
        Intent i = new Intent(ctx, SvitloWidget.class).setAction(ACTION_REFRESH);
        PendingIntent pi = PendingIntent.getBroadcast(ctx, 1, i,
                PendingIntent.FLAG_IMMUTABLE | PendingIntent.FLAG_UPDATE_CURRENT);
        long now = System.currentTimeMillis();
        long when = now + 30 * 60 * 1000L;
        if (change > now && change + 5000 < when) when = change + 5000;
        am.set(AlarmManager.RTC, when, pi);
    }
}
